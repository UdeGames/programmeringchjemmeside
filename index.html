import time
from machine import Pin
import random

redLed = Pin(7, Pin.OUT)
greenLed = Pin(6, Pin.OUT)
blueLed = Pin(5, Pin.OUT)

redButton = Pin(41, Pin.IN, Pin.PULL_UP)
greenButton = Pin(42, Pin.IN, Pin.PULL_UP)
blueButton = Pin(2, Pin.IN, Pin.PULL_UP)

LEDlist = [redLed, greenLed, blueLed]

def ValgLed():
    nrLed = random.randint(0, 2)
    return nrLed

def ButtonPressed():
    while True:
        redButtonStatus = Pin.value(redButton)
        greenButtonStatus = Pin.value(greenButton)
        blueButtonStatus = Pin.value(blueButton)

        if redButtonStatus == 0:
            return 0
        if greenButtonStatus == 0:
            return 1
        if blueButtonStatus == 0:
            return 2

nrLed = ValgLed()
Pin.value(LEDlist[nrLed], 1)
startTid = time.ticks_ms()
knap = ButtonPressed()
print("dd")

if knap == nrLed:
    slutTid = time.ticks_ms()
    print(slutTid - startTid)
else:
    print("forkert")

from machine import Pin
from time import sleep
import neopixel

knap = Pin(11, Pin.IN, Pin.PULL_UP)

antal_led = 16
timing_v6 = (300, 950, 700, 600)

np = neopixel.NeoPixel(
    Pin(10, Pin.OUT),
    antal_led,
    bpp=3,
    timing=timing_v6
)

while True:
    if knap.value() == 0:
        for i in range(antal_led):
            np[i] = (255, 0, 0)
        np.write()
    else:
        np.fill((0,0,0))
        np.write()
    
    sleep(0.1)

Projektbeskrivelse

Vi fik igen udleveret et program klippet i stykker på papir. Programmet var en form for baksensor ligesom jeg selv havde lavet tidelige i programmerings forløbet i en opgave jeg glemte at gemme.

Kildekode / Resultat


import time
from machine import Pin, time_pulse_us 

redLed = Pin(7, Pin.OUT)
greenLed = Pin(6, Pin.OUT)
trig = Pin(12, Pin.OUT)
echo = Pin(13, Pin.IN)

def maalAfstand():
    trig.value(0)
    time.sleep_us(2)
    trig.value(1)
    time.sleep_us(10)
    trig.value(0)

    varighed = time_pulse_us(echo, 1, 30000)
    afstand = varighed / 58
    return afstand

while True:
    afstand = maalAfstand()
    if afstand < 20:
        redLed.value(1)
        greenLed.value(0)
        print("Stop!")
    else:
        redLed.value(0)
        greenLed.value(1)
        print("Fri bane!")
    time.sleep_ms(200)
